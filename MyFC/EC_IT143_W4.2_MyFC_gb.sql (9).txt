USE MyFC;
GO

IF OBJECT_ID('dbo.tblPlayersPerTeam', 'U') IS NOT NULL
    DROP TABLE dbo.tblPlayersPerTeam;
GO

SELECT
    TeamCode,
    PlayerCount
INTO dbo.tblPlayersPerTeam
FROM dbo.vw_MyFC_PlayersPerTeam;
GO
