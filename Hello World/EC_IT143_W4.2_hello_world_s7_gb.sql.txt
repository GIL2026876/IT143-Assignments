NAME: EC_IT143_W4.2_hello_world_s7_gb

PURPOSE:
Convert the loading script into a stored procedure.
*/

USE EC_IT143_DA;
GO

CREATE OR ALTER PROCEDURE dbo.usp_hello_world_load
AS
BEGIN
    SET NOCOUNT ON;

    TRUNCATE TABLE dbo.t_hello_world;

    INSERT INTO dbo.t_hello_world
    (
        my_message,
        current_date_time
    )
    SELECT
        v.my_message,
        v.current_date_time
    FROM dbo.v_hello_world_load AS v;
END;
GO
